// ============================================
// UNIVERSITY ACADEMIC SYSTEM
// Database: PostgreSQL
// Normalization: 4NF
// ============================================

// ============================================
// INFRASTRUCTURE MODULE
// ============================================

Table building {
  id serial [primary key]
  name varchar(100) [not null]
}

Table classroom {
  id serial [primary key]
  name varchar(100) [not null]
  building_id int [not null]
}

// ============================================
// ACADEMIC MODULE
// ============================================

Table faculty {
  id serial [primary key]
  name varchar(100) [not null]
}

Table program {
  id serial [primary key]
  name varchar(100) [not null]
  faculty_id int [not null]
}

Table course {
  id serial [primary key]
  name varchar(100) [not null]
  faculty_id int [not null]
}

Table schedule {
  id serial [primary key]
  day varchar(20) [not null]
  start_time time [not null]
  end_time time [not null]
}

Table course_schedule {
  id serial [primary key]
  course_id int [not null]
  schedule_id int [not null]
  classroom_id int [not null]
  professor_id int [not null]
  capacity int [not null]
}

Table program_course {
  course_id int [not null]
  program_id int [not null]
  curriculum_semester int [not null]

  Indexes {
    (course_id, program_id) [pk]
  }
}

// ============================================
// USER MODULE
// ============================================

Table role {
  id serial [primary key]
  name varchar(50) [not null]
}

Table user {
  id serial [primary key]
  name varchar(100) [not null]
  last_name varchar(100) [not null]
  role_id int [not null]
}

Table student_program {
  program_id int [not null]
  user_id int [not null]
  current_semester int [not null]

  Indexes {
    (program_id, user_id) [pk]
  }
}

Table student_course {
  course_schedule_id int [not null]
  user_id int [not null]

  Indexes {
    (course_schedule_id, user_id) [pk]
  }
}

// ============================================
// REFERENCES - INFRASTRUCTURE MODULE
// ============================================

Ref: classroom.building_id > building.id [delete: restrict, update: cascade]

// ============================================
// REFERENCES - ACADEMIC MODULE
// ============================================

Ref: program.faculty_id > faculty.id [delete: restrict, update: cascade]
Ref: course.faculty_id > faculty.id [delete: restrict, update: cascade]
Ref: course_schedule.course_id > course.id [delete: restrict, update: cascade]
Ref: course_schedule.schedule_id > schedule.id [delete: restrict, update: cascade]
Ref: course_schedule.classroom_id > classroom.id [delete: restrict, update: cascade]
Ref: program_course.course_id > course.id [delete: restrict, update: cascade]
Ref: program_course.program_id > program.id [delete: restrict, update: cascade]

// ============================================
// REFERENCES - USER MODULE
// ============================================

Ref: user.role_id > role.id [delete: restrict, update: cascade]
Ref: student_program.program_id > program.id [delete: restrict, update: cascade]
Ref: student_program.user_id > user.id [delete: restrict, update: cascade]
Ref: student_course.course_schedule_id > course_schedule.id [delete: restrict, update: cascade]
Ref: student_course.user_id > user.id [delete: restrict, update: cascade]
